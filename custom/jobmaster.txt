// Checks if the Player has the required level.
// closes if not, returns if yes
function	Require_Level	{
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;
		mes "Nível necessário:";
		if (getarg(0) > 1) {
			.@temp1$ = "^bb0000"+getarg(0)+"^000000";
			.@base_str$ = .@temp1$ + " (^bb0000Base^000000) / ";
		} else {
			.@base_str$ = "";
		}
		.@temp2$ = "^00bb00"+getarg(1)+"^000000";
		.@job_str$ = .@temp2$ + " (^00bb00Job^000000)";
		mes .@base_str$ + .@job_str$;
		mes "Você precisa de mais ";
		if (.@blvl > 0) {
			.@temp3$ = "^bb0000"+.@blvl+"^000000 níveis de base";
			if (.@jlvl > 0) {
				.@need_base$ = .@temp3$ + " e ";
			} else {
				.@need_base$ = .@temp3$;
			}
		} else {
			.@need_base$ = "";
		}
		if (.@jlvl > 0) {
			.@temp4$ = "^00bb00"+.@jlvl+"^000000 níveis de classe";
			.@need_job$ = .@temp4$;
		} else {
			.@need_job$ = "";
		}
		mes .@need_base$ + .@need_job$ + " para continuar.";
		close;
	}
	return;
}

// Checks if the given eac is an expanded class : Super Novice, Taekwon, Gunslinger, Ninja, Summoner
function Is_Expanded	{
	switch (BaseJob) {
		case JOB_SUPER_NOVICE:
		case JOB_TAEKWON:
		case JOB_STAR_GLADIATOR:
		case JOB_SOUL_LINKER:
		case JOB_GUNSLINGER:
		case JOB_REBELLION:
		case JOB_NINJA:
		case JOB_KAGEROU:
		case JOB_OBORO:
		case JOB_SUMMONER:
			return true;
		default:
			return false;
	}
}

// Checks if the given eac is a baby class
function Is_Baby	{
	return ((getarg(0, eaclass())&EAJL_BABY)>0);
}

// Checks if the player can change to fourth class.
// Note: This does not include the level checks.
function	Can_Change_Fourth	{
	// To change to fourth class you need to be:
	// * Transcendent Third Class
	.@eac = eaclass();

	if( !.TraitClass )
		return false; // Mudança para 4ª classe desativada
	if( .@eac&EAJL_FOURTH )
		return false; // Já é da 4ª Classe
	if( !(.@eac&EAJL_THIRD) )
		return false; // Não é da 3ª Classe
	if( roclass((.@eac&(~EAJL_UPPER))|EAJL_FOURTH) < 0 )
		return false; // A classe não possui 4ª Classe
	if (Is_Expanded())
		return false;
	return true;
}

// Checks if the player can change to third class.
// Note: This does not include the level checks.
function	Can_Change_Third	{
	// To change to third class you either need to be:
	// * Second Class
	// * Transcendent Second Class
	// * Baby Second Class
	.@eac = eaclass();

	if( !.RenewalClass )
		return false; // Mudança para 3ª classe desativada
	if( .@eac&EAJL_THIRD )
		return false; // Já é da 3ª Classe
	if( !(.@eac&EAJL_2) )
		return false; // Não é da 2ª Classe
	if( roclass(.@eac|EAJL_THIRD) < 0 )
		return false; // A classe não possui 3ª Classe
	if( Is_Baby() && (!.BabyClass || !.BabyRenewal) )
		return false; // Mudança para Baby (3ª) não permitida
	if (Is_Expanded())
		return false;
	return true;
}

function	Can_Rebirth	{
	// To rebirth, you need to be:
	// * Second Class
	.@eac = eaclass();

	if( !.RebirthClass )
		return false; // Renascimento desativado
	if( .@eac&(EAJL_UPPER|EAJL_THIRD) )
		return false; // Já renasceu / Já é da 3ª Classe
	if( !(.@eac&EAJL_2) )
		return false; // Não é da 2ª Classe
	if( roclass(.@eac|EAJL_UPPER) < 0 )
		return false; // A classe não possui classe transclasse
	if( Is_Baby() && !.BabyClass )
		return false; // Mudanças para Baby não permitidas
	if (Is_Expanded())
		return false;
	return true;
}

// Checks if the given eac is a first class
// Expanded classes other than Taekwon have their own logic in order to follow the npc setting
//
// Swordman (1st class) => Knight (2nd class) => Rune Knight (3rd class) (renewal class) => Dragon Knight (4th class) (trait class)
// Taekwon (1st class) => Star Gladiator (2nd class) => Star Emperor (3rd class) (renewal class) => Sky Emperor (4th class) (trait class)
//
// Gunslinger (1st class) => Rebellion (2nd class) (renewal class) => Night Watch (3rd class) (trait class)
// Summoner (1st class) => Spirit Handler (2nd class) (trait class)
function	Is_First_Cls	{
	return (getarg(0) <= EAJ_TAEKWON);
}

function	Check_Riding	{
	// Note: Why we should always check for Riding:
	// Mounts are considered as another class, which
	// would make this NPC bigger just to handle with
	// those special cases.
	.@string$ = (checkfalcon() ? "Falcão" : "");
	.@string$ += (checkcart() ? (.@string$ != "" ? ", Carrinho" : "Carrinho") : "");
	.@string$ += (checkriding() ? (.@string$ != "" ? ", Peco Peco" : "Peco Peco") : "");
	.@string$ += (ismounting() ? (.@string$ != "" ? ", Montaria" : "Montaria") : "");
	.@string$ += (checkdragon() ? (.@string$ != "" ? ", Dragão" : "Dragão") : "");
	.@string$ += (checkmadogear() ? (.@string$ != "" ? ", Mado" : "Mado") : "");
	.@string$ += (checkwug() ? (.@string$ != "" ? ", Worg" : "Worg") : "");
	.@string$ += (checkhomcall() == 0 ? (.@string$ != "" ? ", Homunculus" : "Homunculus") : "");

	if (.@string$ != "") {
		mes "Por favor, remova seu/sua " +.@string$ + " antes de prosseguir.";
		close;
	}
	return;
}

function	Check_SkillPoints	{
	if (.SkillPointCheck && SkillPoint) {
		mes "Por favor, utilize todos os seus pontos de habilidade antes de prosseguir.";
		close;
	}
	return;
}

// addJobOptions is essentially the same like
// setarray .@array[getarraysize(.@array)],opt1,opt2,...;
// It's just easier to read, since we're using it very often
function	Job_Options	{
	.@argcount = getargcount();
	.@arr_size = getarraysize(getarg(0));
	for( .@i = 1; .@i < .@argcount; .@i++) {
		setarray getelementofarray(getarg(0), .@arr_size++),getarg(.@i);
	}
}

// Begin of the NPC
	mes .NPCName$;
	Check_Riding();
	Check_SkillPoints();

	// initialisation
	.@eac = eaclass();
	.@fourth_possible = Can_Change_Fourth();
	.@third_possible = Can_Change_Third();
	.@rebirth_possible = Can_Rebirth();
	.@first_eac = .@eac&EAJ_BASEMASK;
	.@second_eac = .@eac&EAJ_UPPERMASK;
	.@nobaby_eac = .@eac&(~EAJL_BABY);
	.@noupper_eac = .@eac&(~EAJL_UPPER);
	// Note: These are already set in pc.cpp
	// BaseClass = roclass(.@eac&EAJ_BASEMASK) which is the players First Class
	// BaseJob = roclass(.@eac&EAJ_UPPERMASK) which is the players Second Class

	// From here on the jobmaster checks the current class
	// and fills the array `.@job_opt` with possible
	// job options for the player.

	if( .@rebirth_possible ) {
		// Rebirth option (displayed on the top of the menu)
		Require_Level(.Req_Rebirth[0], .Req_Rebirth[1]);
		Job_Options(.@job_opt, Job_Novice_High);
	}
	if( .@third_possible ) {
		// Third Job change (displayed below rebirth)
		Require_Level(.Req_Third[0], .Req_Third[1]);
		Job_Options(.@job_opt, roclass(.@eac|EAJL_THIRD));
	}
	if( .@fourth_possible ) {
		// Fourth Job change (displayed below rebirth)
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(.@noupper_eac|EAJL_FOURTH));
	}

	// Whether the character can become renewal (baby) expanded class : Expanded Super Novice
	if (.RenewalExpanded &&
	   (.@eac&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE && // is Super Novice
	   !(eaclass()&EAJL_THIRD) ) {			// not already Expanded SN
		// (Baby) Super Novice to Expanded (Baby) Super Novice
		if( !Is_Baby(.@eac) || (.BabyClass && .BabyExpanded) ) {
			// .BabyClass & .BabyExpanded must be enabled if the is a baby
			Require_Level(.Req_Exp_SNOVI[0], .Req_Exp_SNOVI[1]);
			Job_Options(.@job_opt,roclass(.@eac|EAJL_THIRD)); // Expanded SN is "third" cls
		}
	}

	// babies dont get to become 4th jobs
	if (.TraitExpanded && .@eac == EAJ_REBELLION) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(EAJ_NIGHT_WATCH));	// .@eac|EAJL_THIRD
	}

	if (.TraitExpanded && .@eac == EAJ_KAGEROUOBORO) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(EAJ_SHINKIROSHIRANUI));	// .@eac|EAJL_THIRD
	}

	if (.TraitExpanded && .@eac == EAJ_SUPER_NOVICE_E) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(EAJ_HYPER_NOVICE));	// .@eac|EAJL_FOURTH
	}

	if (.TraitExpanded && .@eac == EAJ_SUMMONER) {
		Require_Level(.Req_SHandler[0], .Req_SHandler[1]);
		Job_Options(.@job_opt, roclass(EAJ_SPIRIT_HANDLER));	// .@eac|EAJL_2_1

		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(EAJ_SKY_EMPEROR));	// .@eac|EAJL_FOURTH
	}

	if (.TraitExpanded && .@eac == EAJ_SOUL_REAPER) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(EAJ_SOUL_ASCETIC));	// .@eac|EAJL_FOURTH
	}

	// Whether the character can become renewal (baby) expanded class - a class above level 99 without trait points : Rebellion, Kagerou/Oboro, Star Emperor, Soul Reaper
	if (.RenewalExpanded) {
		// .BabyClass & .BabyExpanded must be enabled if the is a baby
		if( !Is_Baby(.@eac) || (.BabyClass && .BabyExpanded) ) {
			switch (.@nobaby_eac) {
				case EAJ_SOULLINKER:
				case EAJ_STAR_GLADIATOR:
					// (Baby) Soul Linker to (Baby) Soul Reaper
					// (Baby) Star Gladiator to (Baby) Star Emperor
					// Star Emperor and Soul Reaper are considered as 3rd class
					Require_Level(.Req_Third[0], .Req_Third[1]);
					Job_Options(.@job_opt, roclass(.@eac|EAJL_THIRD));
					break;
				case EAJ_NINJA:
				case EAJ_GUNSLINGER:
					// (Baby) Ninja to (Baby) Kagerou / Oboro
					// (Baby) Gunslinger to (Baby) Rebellion
					// Kagerou, Oboro, Rebellion are considered as 2-1 class
					Require_Level(.Req_Exp_NJ_GS[0], .Req_Exp_NJ_GS[1]);
					Job_Options(.@job_opt, roclass(.@eac|EAJL_2_1));
					break;
			}
		}
	}

	// Classes Personalizadas do evolucao.txt - Summoner -> Druid -> Karnos -> Alitea
	// So aplicavel quando Summoner é uma 2ª classe (nao classe inicial)
	// Requisitos conforme evolucao.txt: Summoner->Druid usa 99/50, Druid->Karnos usa 200/70, Karnos->Alitea usa 200/70
	if (.RenewalClass && BaseJob == Job_Summoner && (.@eac&EAJL_2) && !(.@eac&EAJL_THIRD)) {
		Require_Level(.Req_Third[0], .Req_Third[1]);
		Job_Options(.@job_opt, Job_Druid);

	if (.RenewalClass && BaseJob == Job_Druid && (.@eac&EAJL_2) && !(.@eac&EAJL_THIRD)) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, Job_Karnos);

	if (.TraitClass && BaseJob == Job_Karnos && (.@eac&EAJL_2) && !(.@eac&EAJL_FOURTH)) {
		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, Job_Alitea);
	}

	// Player is Job_Novice, Job_Novice_High or Job_Baby
	// Also handle Summoner as starting class (if server config allows it)
	if ((.@first_eac == EAJ_NOVICE && .@second_eac != EAJ_SUPER_NOVICE) ||

					Job_Options(.@job_opt, Job_Baby);
				break;
			case Job_Novice_High:

				if( .LastJob && lastJob )

					Job_Options(.@job_opt, Job_Swordman_High, Job_Mage_High, Job_Archer_High,Job_Acolyte_High, Job_Merchant_High, Job_Thief_High);
				break;

				if( !.BabyClass )
					break;
				// First job change as a baby
				Job_Options(.@job_opt, Job_Baby_Swordman, Job_Baby_Mage,Job_Baby_Archer,Job_Baby_Acolyte, Job_Baby_Merchant,Job_Baby_Thief);

					Job_Options(.@job_opt, Job_Super_Baby, Job_Baby_Taekwon,Job_Baby_Gunslinger, Job_Baby_Ninja);
				if( .BabySummoner )
					Job_Options(.@job_opt, Job_Baby_Summoner);
				break;
			default:
				mes "Ocorreu um erro.";
				close;
		}
	} else if( Is_First_Cls(.@eac) || 				// First Class
		   Is_First_Cls(.@eac&(~EAJL_UPPER)) ||	// Trans. First Cls
		   (.BabyClass && Is_First_Cls(.@nobaby_eac)) ) {	// Baby First Cls

		// most jobs should have two options here  (2-1 and 2-2)

			Job_Options(.@job_opt, lastJob + Job_Novice_High);
		} else {
			// Class is not enforced, player can decide.
			if( .@class1 > 0 ) { // 2-1
				Require_Level(.Req_Second[0], .Req_Second[1]);
				Job_Options(.@job_opt, .@class1);
			}
			if( .@class2 > 0 ) { // 2-2
				Require_Level(.Req_Second[0], .Req_Second[1]);
				Job_Options(.@job_opt, .@class2);
			}
		}
	}

// Displays the job menu
function	Job_Menu	{
	// getarg(0) is the .@job_opt array holding all available job changes.
	function Confirm_Change;
	while(true) {
		.@opt_cnt =  getarraysize(getarg(0));
		if( .@opt_cnt <= 0 ) {
			mes "Não há mais classes disponíveis.";
			close;
		}

		.@selected = 0; // Just a single job class given, no select needed
		if (.@opt_cnt > 1) {
			// Multiple job classes given. Select one and save it to .@class
			// After that confirm .@class
			mes "Selecione uma classe.";
			.@menu$ = "";
			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if( getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FFRenascimento^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));
				.@menu$ = .@menu$ + "~" + .@jobname$ + ":";
			}
			.@menu$ = .@menu$+"~^777777Cancelar^000000";
			.@selected = select(.@menu$) - 1;
			if( .@selected < 0 || .@selected >= .@opt_cnt )
				close;
			next;
			mes .NPCName$;
		}
		.@class = getelementofarray(getarg(0), .@selected);
		if ((.@class == Job_Super_Novice || .@class == Job_Super_Baby) && BaseLevel < .SNovice) {
			mes "É necessário um nível base de " + .SNovice + " para se tornar um(a) " + jobname(.@class) + ".";
			return;
		}
		// Confirm the Class
		Confirm_Change(.@class, .@opt_cnt > 1);
		next;
		mes .NPCName$;
	}
	return;
}

// Executes the actual jobchange and closes.
function	Job_Change	{
	.@previous_class = Class;
	.@to_cls = getarg(0);
	next;
	mes .NPCName$;
	mes "Agora você é " + callfunc("F_InsertArticle", jobname(.@to_cls)) + "!";
	if (.@to_cls == Job_Novice_High && .LastJob)
		lastJob = Class; // Saves the lastJob for rebirth
	jobchange .@to_cls;
	if (.@to_cls == Job_Novice_High)
		resetlvl(1);
	else if (.@to_cls == Job_Baby) {
		resetstatus;
		resetskill;
		set SkillPoint,0;
	}
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	if (.@previous_class != Class) {
		if (.Platinum)
			callfunc "F_GetPlatinumSkills";
		if (.GetJobEquip)
			Get_Job_Equip();
	}
	close; // Always closes after the change
}

function	Confirm_Change	{
	// Player confirms he want to change into .@class
	.@class = getarg(0, -1);
	.@back = getarg(1, false);
	if( .@class < 0 || eaclass(.@class) == -1 ) {
		mes "Erro de classe desconhecida.";
		close;
	}
	mes "Deseja mudar para a classe ^0055FF"+jobname(.@class)+"^000000?";
	.@job_option$ = " ~ Mudar para a classe ^0055FF"+jobname(.@class)+"^000000";
	if( .@class == Job_Novice_High)
		.@job_option$ = " ~ ^0055FFRenascimento^000000";
	if (select(.@job_option$+":~ ^777777" + ((.@back) ?"Voltar" : "Cancelar") + "^000000") == 1) {
		Job_Change(.@class);
	}
	if (!.@back)
		close; // "Cancel" pressed
	return;
}
